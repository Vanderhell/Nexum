cmake_minimum_required(VERSION 3.16)
project(Nexum VERSION 0.1 LANGUAGES C)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)

add_library(pnp STATIC src/pnp_primitive.c src/pnp_validate.c src/pnp_graph.c src/pnp_analysis.c src/pnp_serialize.c src/nexum_program.c src/nexum_builder.c src/nexum_config.c src/nexum_compile.c)
target_include_directories(pnp PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
if(MSVC)
  target_compile_options(pnp PRIVATE /W4)
else()
  target_compile_options(pnp PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
endif()

enable_testing()
foreach(test_name primitive graph topology determinism expressivity expressivity_regression fsm generated serialization analysis retry freeze cross_toolchain program builder compiler)
  add_executable(test_${test_name} tests/test_${test_name}.c)
  target_link_libraries(test_${test_name} PRIVATE pnp)
  if(MSVC)
    target_compile_options(test_${test_name} PRIVATE /W4)
  else()
    target_compile_options(test_${test_name} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
  endif()
  add_test(NAME ${test_name} COMMAND test_${test_name})
endforeach()

add_executable(pnp_benchmark benchmarks/pnp_benchmark.c)
target_link_libraries(pnp_benchmark PRIVATE pnp)
option(PNP_BUILD_FUZZ "Build portable fuzz entry point" OFF)
if(PNP_BUILD_FUZZ)
  if(CMAKE_C_COMPILER_ID MATCHES "Clang")
    add_executable(pnp_fuzz fuzz/fuzz_config.c)
    target_link_libraries(pnp_fuzz PRIVATE pnp)
    target_compile_options(pnp_fuzz PRIVATE -fsanitize=fuzzer,address,undefined -fno-omit-frame-pointer)
    target_link_options(pnp_fuzz PRIVATE -fsanitize=fuzzer,address,undefined)
  else()
    add_library(pnp_fuzz OBJECT fuzz/fuzz_config.c)
    target_link_libraries(pnp_fuzz PRIVATE pnp)
  endif()
endif()
